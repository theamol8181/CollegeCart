rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if user is admin
    function isAdmin(uid) {
      return get(/databases/$(database)/documents/users/$(uid)).data.role == 'admin';
    }
    
    // Helper to check auth
    function isLoggedIn() {
      return request.auth != null;
    }
    
    // ========== PRODUCTS COLLECTION ==========
    match /products/{productId} {
      // Anyone can READ all products
      allow read: if true;
      
      // Authenticated users can CREATE products
      allow create: if isLoggedIn() 
        && request.resource.data.sellerId == request.auth.uid
        && request.resource.data.name != null
        && request.resource.data.name != '';
      
      // Only seller OR admin can UPDATE/DELETE
      allow update, delete: if isLoggedIn() 
        && (
          request.auth.uid == resource.data.sellerId 
          || isAdmin(request.auth.uid)
        );
    }
    
    // ========== USERS COLLECTION ==========
    match /users/{userId} {
      // Anyone can READ user profiles
      allow read: if true;
      
      // Users can only WRITE their own profile
      allow write: if isLoggedIn() && request.auth.uid == userId;
    }
    
    // ========== WISHLISTS COLLECTION ==========
    match /wishlists/{wishlistId} {
      // Authenticated users can READ
      allow read: if isLoggedIn();
      
      // Authenticated users can WRITE their own wishlists
      allow write: if isLoggedIn() 
        && request.auth.uid == request.resource.data.userId;
    }
  }
}
